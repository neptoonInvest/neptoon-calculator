function buildPortCustomMetricsHistory(levelHoldings){ const BASE_DATE = "2014-12-31"; const instruments = INSTRUMENT_STATE.all || []; const holdings = (levelHoldings || []).map(h => { const tickerExchange = String(h.tickerExchange || "") .trim() .toUpperCase(); const classNo = Number(h.classNo); const weight = Number(h.weight); const instrument = instruments.find(x => String(x.tickerExchange || "") .trim() .toUpperCase() === tickerExchange && Number(x.classNo) === classNo ); if(!instrument){ throw new Error( `Custom portfolio: instrument not found ${tickerExchange}` ); } if(!Number.isFinite(weight) || weight < 0){ throw new Error( `Invalid weight: ${tickerExchange}` ); } return { ...instrument, weight }; }); if(!holdings.length){ throw new Error("Custom portfolio has no holdings"); } const totalWeight = holdings.reduce( (sum, h) => sum + h.weight, 0 ); if(Math.abs(totalWeight - 1) > 0.000001){ throw new Error( `Custom portfolio weights must sum to 1: ${totalWeight}` ); } /* ========================================== 1 · PORTFOLIO METRICS ========================================== */ const objKeyFigures = getMetrics(holdings); /* ========================================== 2 · CANONICAL MONTHLY DATES ========================================== */ const historyMaps = holdings.map(h => { const history = INSTRUMENT_HISTORY_STATE.byTicker.get( String(h.tickerExchange || "") .trim() .toUpperCase() ) || []; const map = new Map(); history.forEach(row => { const date = String(row.date || "").slice(0, 10); const index = Number(row.accIndexTotal); if( date && Number.isFinite(index) && index > 0 ){ map.set(date, index); } }); if(!map.has(BASE_DATE)){ throw new Error( `Missing base date history: ${h.tickerExchange}` ); } return map; }); // Use the first instrument's monthly dates // as the canonical timeline. const canonicalDates = [...historyMaps[0].keys()] .filter(date => date >= BASE_DATE) .sort(); if(canonicalDates[0] !== BASE_DATE){ throw new Error( "Custom portfolio base date is missing" ); } /* ========================================== 3 · CONSTANT-WEIGHT MONTHLY RETURNS ========================================== */ let nav = 100; const arrayHistory = [ { date: BASE_DATE, accPortIndex: 100 } ]; for(let t = 1; t < canonicalDates.length; t++){ const date = canonicalDates[t]; const previousDate = canonicalDates[t - 1]; let portfolioReturn = 0; holdings.forEach((holding, i) => { const map = historyMaps[i]; const previous = map.get(previousDate); const current = map.get(date); if( !Number.isFinite(previous) || !Number.isFinite(current) || previous <= 0 ){ throw new Error( `Missing monthly history: ${holding.tickerExchange} ${date}` ); } const monthlyReturn = current / previous - 1; portfolioReturn += holding.weight * monthlyReturn; }); nav *= 1 + portfolioReturn; arrayHistory.push({ date, accPortIndex: nav }); } /* ========================================== 4 · RETURN PORTFOLIO COMPONENTS ========================================== */ return { arrayHoldings: holdings, objKeyFigures, arrayHistory }; }
